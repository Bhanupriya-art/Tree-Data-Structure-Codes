// Java - Segment Tree for Range Sum Query and Point Update
class SegmentTree {
    int[] seg, arr;
    int n;

    SegmentTree(int[] input) {
        n = input.length;
        arr = input.clone();
        seg = new int[4 * n];
        build(0, 0, n - 1);
    }

    void build(int idx, int l, int r) {
        if (l == r) {
            seg[idx] = arr[l];
            return;
        }
        int mid = (l + r) / 2;
        build(2 * idx + 1, l, mid);
        build(2 * idx + 2, mid + 1, r);
        seg[idx] = seg[2 * idx + 1] + seg[2 * idx + 2];
    }

    void update(int idx, int l, int r, int pos, int val) {
        if (l == r) {
            seg[idx] = val;
            return;
        }
        int mid = (l + r) / 2;
        if (pos <= mid)
            update(2 * idx + 1, l, mid, pos, val);
        else
            update(2 * idx + 2, mid + 1, r, pos, val);
        seg[idx] = seg[2 * idx + 1] + seg[2 * idx + 2];
    }

    int query(int idx, int l, int r, int ql, int qr) {
        if (ql > r || qr < l) return 0;
        if (ql <= l && qr >= r) return seg[idx];
        int mid = (l + r) / 2;
        return query(2 * idx + 1, l, mid, ql, qr) +
               query(2 * idx + 2, mid + 1, r, ql, qr);
    }

    public static void main(String[] args) {
        int[] arr = {1, 3, 5, 7, 9, 11};
        SegmentTree st = new SegmentTree(arr);
        System.out.println(st.query(0, 0, arr.length - 1, 1, 3)); // 3+5+7=15
        st.update(0, 0, arr.length - 1, 1, 10); // arr[1] = 10
        System.out.println(st.query(0, 0, arr.length - 1, 1, 3)); // 10+5+7=22
    }
}
# Python - Segment Tree for Range Sum and Point Update
class SegmentTree:
    def __init__(self, arr):
        self.n = len(arr)
        self.seg = [0] * (4 * self.n)
        self.build(arr, 0, 0, self.n - 1)

    def build(self, arr, idx, l, r):
        if l == r:
            self.seg[idx] = arr[l]
            return
        mid = (l + r) // 2
        self.build(arr, 2 * idx + 1, l, mid)
        self.build(arr, 2 * idx + 2, mid + 1, r)
        self.seg[idx] = self.seg[2 * idx + 1] + self.seg[2 * idx + 2]

    def update(self, idx, l, r, pos, val):
        if l == r:
            self.seg[idx] = val
            return
        mid = (l + r) // 2
        if pos <= mid:
            self.update(2 * idx + 1, l, mid, pos, val)
        else:
            self.update(2 * idx + 2, mid + 1, r, pos, val)
        self.seg[idx] = self.seg[2 * idx + 1] + self.seg[2 * idx + 2]

    def query(self, idx, l, r, ql, qr):
        if ql > r or qr < l:
            return 0
        if ql <= l and qr >= r:
            return self.seg[idx]
        mid = (l + r) // 2
        return self.query(2 * idx + 1, l, mid, ql, qr) + \
               self.query(2 * idx + 2, mid + 1, r, ql, qr)

// C++ - Segment Tree for Range Sum Query and Point Update
#include <iostream>
#include <vector>
using namespace std;

class SegmentTree {
    vector<int> seg;
    int n;

    void build(vector<int>& arr, int idx, int l, int r) {
        if (l == r) {
            seg[idx] = arr[l];
            return;
        }
        int mid = (l + r) / 2;
        build(arr, 2 * idx + 1, l, mid);
        build(arr, 2 * idx + 2, mid + 1, r);
        seg[idx] = seg[2 * idx + 1] + seg[2 * idx + 2];
    }

    void update(int idx, int l, int r, int pos, int val) {
        if (l == r) {
            seg[idx] = val;
            return;
        }
        int mid = (l + r) / 2;
        if (pos <= mid)
            update(2 * idx + 1, l, mid, pos, val);
        else
            update(2 * idx + 2, mid + 1, r, pos, val);
        seg[idx] = seg[2 * idx + 1] + seg[2 * idx + 2];
    }

    int query(int idx, int l, int r, int ql, int qr) {
        if (ql > r || qr < l) return 0;
        if (ql <= l && qr >= r) return seg[idx];
        int mid = (l + r) / 2;
        return query(2 * idx + 1, l, mid, ql, qr) +
               query(2 * idx + 2, mid + 1, r, ql, qr);
    }

public:
    SegmentTree(vector<int>& arr) {
        n = arr.size();
        seg.resize(4 * n);
        build(arr, 0, 0, n - 1);
    }

    void update(int pos, int val) {
        update(0, 0, n - 1, pos, val);
    }

    int query(int l, int r) {
        return query(0, 0, n - 1, l, r);
    }
};

int main() {
    vector<int> arr = {1, 3, 5, 7, 9, 11};
    SegmentTree st(arr);
    cout << st.query(1, 3) << endl; // 15
    st.update(1, 10);               // arr[1] = 10
    cout << st.query(1, 3) << endl; // 22
    return 0;
}
