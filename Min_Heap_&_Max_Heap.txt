// Java - MinHeap and MaxHeap using ArrayList
import java.util.*;

class MinHeap {
    List<Integer> heap = new ArrayList<>();

    void insert(int val) {
        heap.add(val);
        int i = heap.size() - 1;
        while (i > 0 && heap.get(i) < heap.get((i - 1) / 2)) {
            Collections.swap(heap, i, (i - 1) / 2);
            i = (i - 1) / 2;
        }
    }

    void printHeap() {
        System.out.println("Min-Heap: " + heap);
    }
}

class MaxHeap {
    List<Integer> heap = new ArrayList<>();

    void insert(int val) {
        heap.add(val);
        int i = heap.size() - 1;
        while (i > 0 && heap.get(i) > heap.get((i - 1) / 2)) {
            Collections.swap(heap, i, (i - 1) / 2);
            i = (i - 1) / 2;
        }
    }

    void printHeap() {
        System.out.println("Max-Heap: " + heap);
    }
}

public class HeapExample {
    public static void main(String[] args) {
        MinHeap minHeap = new MinHeap();
        MaxHeap maxHeap = new MaxHeap();
        int[] values = {10, 4, 15, 20, 0, 7};

        for (int v : values) {
            minHeap.insert(v);
            maxHeap.insert(v);
        }

        minHeap.printHeap(); // Min-Heap
        maxHeap.printHeap(); // Max-Heap
    }
}

    heapq.heappush(min_heap, v)

// C++ - MinHeap and MaxHeap using priority_queue
#include <iostream>
#include <queue>
#include <vector>
using namespace std;

int main() {
    vector<int> values = {10, 4, 15, 20, 0, 7};

    // Min-Heap
    priority_queue<int, vector<int>, greater<int>> minHeap;
    for (int v : values) minHeap.push(v);
    cout << "Min-Heap: ";
    while (!minHeap.empty()) {
        cout << minHeap.top() << " ";
        minHeap.pop();
    }
    cout << endl;

    // Max-Heap
    priority_queue<int> maxHeap;
    for (int v : values) maxHeap.push(v);
    cout << "Max-Heap: ";
    while (!maxHeap.empty()) {
        cout << maxHeap.top() << " ";
        maxHeap.pop();
    }
    cout << endl;

    return 0;
}
